## 2. 웨어하우스 (Warehouse)
```sql
CREATE WAREHOUSE MY_WH
  WAREHOUSE_SIZE = 'XSMALL'
  AUTO_SUSPEND = 60
  AUTO_RESUME = TRUE
  INITIALLY_SUSPENDED = TRUE;

SHOW WAREHOUSES;
ALTER WAREHOUSE MY_WH SET WAREHOUSE_SIZE = 'SMALL';
ALTER WAREHOUSE MY_WH SUSPEND;

```

## 3. 데이터베이스 / 스키마

## 4. 테이블
```sql
-- 참고: PRIMARY KEY / UNIQUE / FOREIGN KEY는 메타데이터일 뿐 강제되지 않음 (NOT NULL만 강제, 일반 테이블 기준 — Hybrid Table은 PK/UNIQUE/FK 강제)
CREATE TABLE CUSTOMERS (
  ID         NUMBER AUTOINCREMENT PRIMARY KEY,
  NAME       VARCHAR(100) NOT NULL,
  EMAIL      VARCHAR(200),
  CREATED_AT TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP()
);

SHOW TABLES;
SHOW TABLES LIKE 'CUST%' IN SCHEMA MY_DB.MY_SCHEMA;
DESCRIBE TABLE CUSTOMERS;

-- INFORMATION_SCHEMA로 메타데이터 조회
SELECT TABLE_NAME, ROW_COUNT, BYTES FROM MY_DB.INFORMATION_SCHEMA.TABLES WHERE TABLE_SCHEMA = 'MY_SCHEMA';
SELECT COLUMN_NAME, DATA_TYPE FROM MY_DB.INFORMATION_SCHEMA.COLUMNS WHERE TABLE_NAME = 'CUSTOMERS' ORDER BY ORDINAL_POSITION;

ALTER TABLE CUSTOMERS ADD COLUMN PHONE VARCHAR(20);
ALTER TABLE CUSTOMERS RENAME COLUMN PHONE TO MOBILE;
ALTER TABLE CUSTOMERS DROP COLUMN MOBILE;

CREATE TABLE CUSTOMERS_BAK CLONE CUSTOMERS;      -- 제로카피 클론
CREATE TABLE CUSTOMERS_CTAS AS SELECT * FROM CUSTOMERS;  -- CTAS: DEFAULT/AUTOINCREMENT/제약조건은 복사 안 됨
CREATE TABLE CUSTOMERS_EMPTY LIKE CUSTOMERS;             -- 구조만 복사 (데이터 없음)

TRUNCATE TABLE CUSTOMERS;
DROP TABLE CUSTOMERS;
UNDROP TABLE CUSTOMERS;

-- 테이블 이름 변경 (이후 예제는 CUSTOMERS 이름을 계속 사용하므로 맨 마지막에 둠)
ALTER TABLE CUSTOMERS RENAME TO CLIENTS;
```

## 5. DML (데이터 조작)
```sql
INSERT INTO CUSTOMERS (NAME, EMAIL) VALUES ('홍길동', 'hong@example.com');

UPDATE CUSTOMERS SET EMAIL = 'new@example.com' WHERE ID = 1;

DELETE FROM CUSTOMERS WHERE ID = 1;

-- 주의: AUTOINCREMENT 컬럼(ID)에 값을 직접 넣으면 이후 자동 채번 값과 중복될 수 있음
MERGE INTO CUSTOMERS t
USING STAGING_CUSTOMERS s ON t.ID = s.ID
WHEN MATCHED THEN UPDATE SET t.NAME = s.NAME, t.EMAIL = s.EMAIL
WHEN NOT MATCHED THEN INSERT (ID, NAME, EMAIL) VALUES (s.ID, s.NAME, s.EMAIL);
```

## 6. 조회 (SELECT)
```sql
SELECT * FROM CUSTOMERS LIMIT 10;

SELECT CUSTOMER_ID, COUNT(*) AS CNT
FROM ORDERS
WHERE ORDER_DATE >= '2026-01-01'
GROUP BY CUSTOMER_ID
HAVING COUNT(*) > 5
ORDER BY CNT DESC;

-- 윈도우 함수 + QUALIFY
SELECT *
FROM ORDERS
QUALIFY ROW_NUMBER() OVER (PARTITION BY CUSTOMER_ID ORDER BY ORDER_DATE DESC) = 1;

-- Time Travel
SELECT * FROM CUSTOMERS AT (OFFSET => -60*5);          -- 5분 전
SELECT * FROM CUSTOMERS BEFORE (STATEMENT => '<query_id>');
```

## 8. 스테이지 & 데이터 로딩
> PUT / GET은 Snowsight 워크시트에서 실행 불가 — SnowSQL, Snowflake CLI(`snow sql`), 드라이버에서 사용

```sql
-- 파일 포맷
CREATE FILE FORMAT MY_CSV
  TYPE = CSV
  FIELD_DELIMITER = ','
  SKIP_HEADER = 1
  FIELD_OPTIONALLY_ENCLOSED_BY = '"';

-- 내부 스테이지
CREATE STAGE MY_STAGE FILE_FORMAT = MY_CSV;

-- 로컬 파일 업로드 (SnowSQL / Snowflake CLI)
PUT 'file://C:/data/customers.csv' @MY_STAGE AUTO_COMPRESS = TRUE;

LIST @MY_STAGE;

-- 적재 전 검증 (실제 적재 없이 오류만 확인)
COPY INTO CUSTOMERS FROM @MY_STAGE/customers.csv.gz
FILE_FORMAT = (FORMAT_NAME = MY_CSV)
VALIDATION_MODE = RETURN_ERRORS;

-- 테이블로 적재
COPY INTO CUSTOMERS
FROM @MY_STAGE/customers.csv.gz
FILE_FORMAT = (FORMAT_NAME = MY_CSV)
ON_ERROR = 'CONTINUE';

-- 테이블 -> 스테이지 언로드
COPY INTO @MY_STAGE/export/ FROM CUSTOMERS FILE_FORMAT = (TYPE = PARQUET);

-- 파일 다운로드
GET @MY_STAGE/export/ 'file://C:/data/export/';

REMOVE @MY_STAGE/customers.csv.gz;
```

## 9. 반정형 데이터 (JSON)

CREATE TABLE RAW_JSON (V VARIANT);

SELECT
  V:name::STRING      AS NAME,
  V:address.city::STRING AS CITY,
  f.value:sku::STRING AS SKU
FROM RAW_JSON,
LATERAL FLATTEN(INPUT => V:items) f;

## 10. 사용자 / 역할 / 권한
```sql
CREATE ROLE ANALYST;
CREATE USER KIM
  PASSWORD = '********'
  MUST_CHANGE_PASSWORD = TRUE
  DEFAULT_ROLE = ANALYST
  DEFAULT_WAREHOUSE = COMPUTE_WH;
-- 사람 계정은 MFA 권장
-- 서비스 계정은 TYPE = SERVICE 사용자 + 키페어 인증(RSA_PUBLIC_KEY) 권장
-- CREATE USER ETL_SVC TYPE = SERVICE RSA_PUBLIC_KEY = 'MIIBIjANBgkqh...' DEFAULT_ROLE = ANALYST;

GRANT ROLE ANALYST TO USER KIM;
GRANT ROLE ANALYST TO ROLE SYSADMIN;   -- 역할 계층

GRANT USAGE ON WAREHOUSE COMPUTE_WH TO ROLE ANALYST;
GRANT USAGE ON DATABASE MY_DB TO ROLE ANALYST;
GRANT USAGE ON SCHEMA MY_DB.MY_SCHEMA TO ROLE ANALYST;
GRANT SELECT ON ALL TABLES IN SCHEMA MY_DB.MY_SCHEMA TO ROLE ANALYST;
GRANT SELECT ON FUTURE TABLES IN SCHEMA MY_DB.MY_SCHEMA TO ROLE ANALYST;
-- 주의: 스키마 레벨 FUTURE GRANT가 있으면 같은 스키마에선 DB 레벨 FUTURE GRANT보다 우선 적용

REVOKE SELECT ON ALL TABLES IN SCHEMA MY_DB.MY_SCHEMA FROM ROLE ANALYST;

SHOW ROLES;
SHOW USERS;
SHOW GRANTS TO ROLE ANALYST;
SHOW GRANTS TO USER KIM;
SHOW GRANTS ON TABLE CUSTOMERS;
```

## 11. 태스크 & 스트림
```sql
-- 스트림 (CDC)
CREATE STREAM CUSTOMERS_STREAM ON TABLE CUSTOMERS;
SELECT * FROM CUSTOMERS_STREAM;   -- SELECT만으로는 스트림이 소비되지 않음 (DML 안에서 읽어야 오프셋 이동)

-- 태스크 (스케줄 작업)
CREATE TASK MY_TASK
  WAREHOUSE = COMPUTE_WH
  SCHEDULE = 'USING CRON 0 9 * * * Asia/Seoul'
  WHEN SYSTEM$STREAM_HAS_DATA('CUSTOMERS_STREAM')
AS
  -- 스트림에는 METADATA$ACTION / METADATA$ISUPDATE / METADATA$ROW_ID 컬럼이 추가되므로 SELECT * 금지
  INSERT INTO CUSTOMERS_HISTORY (ID, NAME, EMAIL, CREATED_AT)
  SELECT ID, NAME, EMAIL, CREATED_AT FROM CUSTOMERS_STREAM;

-- 태스크는 SUSPENDED 상태로 생성됨 → RESUME 필요
-- 권한: 태스크에 대한 OWNERSHIP 또는 OPERATE + 태스크 소유 역할에 계정 레벨 EXECUTE TASK
ALTER TASK MY_TASK RESUME;
ALTER TASK MY_TASK SUSPEND;
EXECUTE TASK MY_TASK;
SHOW TASKS;
```

  TARGET_LAG = '1 hour'

## 13. 함수 & 프로시저
```sql
-- SQL UDF
-- NUMBER만 쓰면 scale 0이라 입력/결과 모두 반올림됨 (예: 15.5 → 16 → 17.6 → 18)
CREATE OR REPLACE FUNCTION ADD_TAX(PRICE NUMBER(38,2))
RETURNS NUMBER(38,2)
AS 'PRICE * 1.1';

-- SQL 저장 프로시저 (변수는 SQL 안에서 :변수명 사용)
CREATE OR REPLACE PROCEDURE DELETE_OLD(DAYS NUMBER)
RETURNS STRING
LANGUAGE SQL
AS
$$
BEGIN
  DELETE FROM ORDERS WHERE ORDER_DATE < DATEADD(DAY, -:DAYS, CURRENT_DATE());
  RETURN 'done';
END;
$$;

CALL DELETE_OLD(30);
```

## 14. 리소스 모니터 (크레딧 한도)
```sql
-- ACCOUNTADMIN 필요
CREATE RESOURCE MONITOR MONTHLY_LIMIT
  WITH CREDIT_QUOTA = 100
  FREQUENCY = MONTHLY
  START_TIMESTAMP = IMMEDIATELY
  TRIGGERS
    ON 80 PERCENT DO NOTIFY          -- 알림 수신 사용자 설정이 별도로 필요
    ON 100 PERCENT DO SUSPEND
    ON 110 PERCENT DO SUSPEND_IMMEDIATE;

ALTER WAREHOUSE COMPUTE_WH SET RESOURCE_MONITOR = MONTHLY_LIMIT;
SHOW RESOURCE MONITORS;
```

## 15. 모니터링 / 기타 유용한 명령
```sql
-- 최근 쿼리 이력 (현재 데이터베이스가 지정되어 있어야 함)
SELECT QUERY_ID, QUERY_TEXT, EXECUTION_STATUS, TOTAL_ELAPSED_TIME
FROM TABLE(INFORMATION_SCHEMA.QUERY_HISTORY())
ORDER BY START_TIME DESC LIMIT 20;

-- 웨어하우스 크레딧 사용량 (최근 7일) — ACCOUNT_USAGE는 최대 수 시간 지연, ACCOUNTADMIN 등 권한 필요
SELECT WAREHOUSE_NAME, SUM(CREDITS_USED) AS CREDITS
FROM SNOWFLAKE.ACCOUNT_USAGE.WAREHOUSE_METERING_HISTORY
WHERE START_TIME >= DATEADD(DAY, -7, CURRENT_TIMESTAMP())
GROUP BY WAREHOUSE_NAME;

-- 실행 중인 쿼리 취소
SELECT SYSTEM$CANCEL_QUERY('<query_id>');
SELECT SYSTEM$CANCEL_ALL_QUERIES(<session_id>);
ALTER USER KIM ABORT ALL QUERIES;

-- 직전 쿼리 결과 재사용
SELECT * FROM TABLE(RESULT_SCAN(LAST_QUERY_ID()));

-- 객체 DDL 확인
SELECT GET_DDL('TABLE', 'CUSTOMERS');

-- 세션 파라미터
ALTER SESSION SET TIMEZONE = 'Asia/Seoul';
SHOW PARAMETERS IN SESSION;
```
